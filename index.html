Skema Tabel SQL Supabase

Jalankan script SQL ini di **SQL Editor** Supabase Anda untuk membuat tabel yang siap menampung 70.000 SKU & hasil Stock Opname: 

-- 1. Tabel Master Item Produk (Di-index untuk 70k SKU)
CREATE TABLE IF NOT EXISTS public.products (
    id BIGSERIAL PRIMARY KEY,
    kode_stock VARCHAR(100) UNIQUE NOT NULL,
    original_code VARCHAR(100),
    nama_barang TEXT,
    family VARCHAR(100),
    qty_akhir NUMERIC DEFAULT 0,
    harga_beli NUMERIC DEFAULT 0,
    harga_cash NUMERIC DEFAULT 0,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_products_kode_stock ON public.products(kode_stock);
CREATE INDEX IF NOT EXISTS idx_products_original_code ON public.products(original_code);

-- 2. Tabel Hasil Stok Opname
CREATE TABLE IF NOT EXISTS public.stock_opname_logs (
    id VARCHAR(100) PRIMARY KEY,
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    petugas VARCHAR(150),
    peran VARCHAR(50),
    rak VARCHAR(100),
    kode_stock VARCHAR(100),
    original_code VARCHAR(100),
    nama_barang TEXT,
    family VARCHAR(100),
    qty_fisik NUMERIC DEFAULT 0
);

-- 3. Matikan Row Level Security (RLS) untuk kemudahan akses API anon
ALTER TABLE public.products DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.stock_opname_logs DISABLE ROW LEVEL SECURITY;

Kode Apps Script Auto-Sync GSheet

Pasang kode script ini di Google Spreadsheet Anda agar setiap hasil scan Stok Opname otomatis tercatat di Sheet **Stok Opname**: 

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    var record = data.record ? data.record : data;
    var id = record.id ? String(record.id) : '';

    // Catat ke Sheet Stok Opname
    var sheetSO = ss.getSheetByName("Stok Opname");
    if (!sheetSO) {
      sheetSO = ss.insertSheet("Stok Opname");
      sheetSO.appendRow(["ID SO", "Waktu", "Petugas", "Peran Petugas", "Rak", "Kode Item", "Kode Original", "Nama Produk", "Family", "Qty"]);
    }
    
    // Cek apakah ID SO sudah ada di Sheet (Mencegah Data Double)
    if (id !== '') {
      var lastRow = sheetSO.getLastRow();
      if (lastRow > 1) {
        var existingIds = sheetSO.getRange(2, 1, lastRow - 1, 1).getValues();
        for (var i = 0; i < existingIds.length; i++) {
          if (String(existingIds[i][0]) === id) {
            return ContentService.createTextOutput(JSON.stringify({"result": "skipped_duplicate", "id": id}))
              .setMimeType(ContentService.MimeType.JSON);
          }
        }
      }
    }

    var qtyVal = record.qty_fisik !== undefined ? record.qty_fisik : (record.qty !== undefined ? record.qty : 0);

    sheetSO.appendRow([
      "'" + id,
      record.timestamp || new Date().toLocaleString('id-ID'),
      record.petugas || '-',
      record.peran || '-',
      record.rak || '-',
      "'" + (record.kode_stock || ''),
      "'" + (record.original_code || ''),
      record.nama_barang || '-',
      record.family || '-',
      qtyVal
    ]);

    return ContentService.createTextOutput(JSON.stringify({"result": "success", "id": id}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
