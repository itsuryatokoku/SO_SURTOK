Skema Tabel SQL Supabase

Jalankan script SQL ini di **SQL Editor** Supabase Anda untuk membuat tabel yang siap menampung 70.000 SKU & hasil Stock Opname: 

-- 1. Tabel Master Item Produk (Di-index untuk 70k SKU)
CREATE TABLE IF NOT EXISTS public.products (
    id BIGSERIAL PRIMARY KEY,
    kode_stock VARCHAR(100) UNIQUE NOT NULL,
    original_code VARCHAR(100),
    nama_barang TEXT,
    family VARCHAR(100),
    qty_akhir NUMERIC DEFAULT 0,
    harga_beli NUMERIC DEFAULT 0,
    harga_cash NUMERIC DEFAULT 0,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_products_kode_stock ON public.products(kode_stock);
CREATE INDEX IF NOT EXISTS idx_products_original_code ON public.products(original_code);

-- 2. Tabel Hasil Stok Opname
CREATE TABLE IF NOT EXISTS public.stock_opname_logs (
    id VARCHAR(100) PRIMARY KEY,
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    petugas VARCHAR(150),
    peran VARCHAR(50),
    rak VARCHAR(100),
    kode_stock VARCHAR(100),
    original_code VARCHAR(100),
    nama_barang TEXT,
    family VARCHAR(100),
    qty_system NUMERIC DEFAULT 0,
    qty_fisik NUMERIC DEFAULT 0,
    selisih NUMERIC DEFAULT 0,
    harga_beli NUMERIC DEFAULT 0,
    harga_cash NUMERIC DEFAULT 0,
    total_nilai_cash NUMERIC DEFAULT 0,
    status VARCHAR(50)
);

-- 3. Matikan Row Level Security (RLS) untuk kemudahan akses API anon
ALTER TABLE public.products DISABLE ROW LEVEL SECURITY;
ALTER TABLE public.stock_opname_logs DISABLE ROW LEVEL SECURITY;

Kode Apps Script Auto-Sync GSheet

Pasang kode script ini di Google Spreadsheet Anda agar setiap hasil scan Stok Opname otomatis tercatat di Sheet **Stok Opname** & **Revisi SO**: 

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    
    // 1. Catat ke Sheet Stok Opname
    var sheetSO = ss.getSheetByName("Stok Opname");
    if (!sheetSO) {
      sheetSO = ss.insertSheet("Stok Opname");
      sheetSO.appendRow(["Waktu", "Petugas", "Peran", "Nama Rak", "Kode Stock", "Original Code", "Nama Barang", "Family", "Qty System", "Qty Fisik", "Selisih", "Harga Beli", "Harga Cash", "Total Cash Fisik", "Status"]);
    }
    
    sheetSO.appendRow([
      data.timestamp || new Date().toLocaleString('id-ID'),
      data.petugas,
      data.peran,
      data.rak,
      "'" + data.kode_stock,
      "'" + data.original_code,
      data.nama_barang,
      data.family,
      data.qty_system,
      data.qty_fisik,
      data.selisih,
      data.harga_beli,
      data.harga_cash,
      data.total_nilai_cash,
      data.status
    ]);

    // 2. Jika ada selisih, catat juga ke Sheet Revisi SO
    if (data.selisih && Number(data.selisih) !== 0) {
      var sheetRevisi = ss.getSheetByName("Revisi SO");
      if (!sheetRevisi) {
        sheetRevisi = ss.insertSheet("Revisi SO");
        sheetRevisi.appendRow(["Waktu", "Petugas", "Nama Rak", "Kode Stock", "Original Code", "Nama Barang", "Family", "Qty System", "Qty Fisik", "Selisih", "Status Selisih"]);
      }
      sheetRevisi.appendRow([
        data.timestamp || new Date().toLocaleString('id-ID'),
        data.petugas,
        data.rak,
        "'" + data.kode_stock,
        "'" + data.original_code,
        data.nama_barang,
        data.family,
        data.qty_system,
        data.qty_fisik,
        data.selisih,
        data.status
      ]);
    }

    return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
