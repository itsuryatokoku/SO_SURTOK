Kode Apps Script Auto-Sync GSheet

Pasang kode script ini di Google Spreadsheet Anda agar pencatatan **Hasil SO** dan update **Revisi SO** berjalan otomatis: 

function doPost(e) {
  var lock = LockService.getScriptLock();
  var success = lock.tryLock(15000);

  if (!success) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": "Lock timeout"}))
      .setMimeType(ContentService.MimeType.JSON);
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var contents = (e && e.postData && e.postData.contents) ? e.postData.contents : null;
    if (!contents) {
      return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": "No payload"}))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var data = JSON.parse(contents);
    var record = data.record ? data.record : data;

    // 1. UPDATE KHUSUS MODE REVISI SO (HANYA UPDATE KOLOM G / QTY REVISI DI SHEET 'Revisi SO')
    var isRevisi = (data.action === "update_revisi" || record.action === "update_revisi" || data.qty_revisi !== undefined || record.qty_revisi !== undefined);

    if (isRevisi) {
      var sheets = ss.getSheets();
      var sheetRevisi = null;
      for (var i = 0; i < sheets.length; i++) {
        if (sheets[i].getName().trim().toLowerCase() === "revisi so") {
          sheetRevisi = sheets[i];
          break;
        }
      }

      if (!sheetRevisi) {
        return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": "Sheet 'Revisi SO' tidak ditemukan"}))
          .setMimeType(ContentService.MimeType.JSON);
      }

      var kodeItem = String(record.kode_stock || record.kode_item || data.kode_stock || data.kode_item || '').replace(/^'/, '').trim();
      if (!kodeItem) {
        return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": "Kode Item kosong"}))
          .setMimeType(ContentService.MimeType.JSON);
      }

      var qtyRev = record.qty_revisi !== undefined ? record.qty_revisi : (data.qty_revisi !== undefined ? data.qty_revisi : record.qty);

      var targetRow = -1;
      var lastRow = sheetRevisi.getLastRow();
      if (lastRow >= 1) {
        var colAValues = sheetRevisi.getRange(1, 1, lastRow, 1).getDisplayValues();
        for (var r = 0; r < colAValues.length; r++) {
          var valInCell = String(colAValues[r][0] || '').replace(/^'/, '').trim();
          if (valInCell.toLowerCase() === kodeItem.toLowerCase()) {
            targetRow = r + 1;
            break;
          }
        }
      }

      if (targetRow > 0) {
        sheetRevisi.getRange(targetRow, 7).setValue(qtyRev); // Tulis HANYA ke Kolom G (QTY Revisi)
        SpreadsheetApp.flush();
        return ContentService.createTextOutput(JSON.stringify({"result": "success", "action": "revisi_updated", "row": targetRow, "qty_revisi": qtyRev}))
          .setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService.createTextOutput(JSON.stringify({"result": "not_found", "message": "Kode Item " + kodeItem + " tidak ditemukan di Sheet Revisi SO"}))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    // 2. HANYA UNTUK STOCK OPNAME FISIK BIASA (SHEET HASIL SO)
    var sheetSO = ss.getSheetByName("Hasil SO");
    if (!sheetSO) {
      sheetSO = ss.insertSheet("Hasil SO");
      sheetSO.appendRow(["ID SO", "Time", "Petugas", "peran Petugas", "Rak", "Kode Item", "Kode Original", "Nama Produk", "Family", "qty"]);
      SpreadsheetApp.flush();
    }

    var id = "";
    if (record.id !== undefined && record.id !== null) {
      id = String(record.id).replace(/^'/, '').trim();
    }

    if (id !== "") {
      var finder = sheetSO.getRange("A:A").createTextFinder(id).matchEntireCell(true);
      var matches = finder.findAll();
      if (matches && matches.length > 0) {
        return ContentService.createTextOutput(JSON.stringify({"result": "skipped_duplicate", "id": id}))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    var qtyVal = record.qty_fisik !== undefined ? record.qty_fisik : (record.qty !== undefined ? record.qty : 0);

    sheetSO.appendRow([
      "'" + id,
      record.timestamp || new Date().toLocaleString('id-ID'),
      record.petugas || '-',
      record.peran || '-',
      record.rak || '-',
      "'" + (record.kode_stock || ''),
      "'" + (record.original_code || ''),
      record.nama_barang || '-',
      record.family || '-',
      qtyVal
    ]);

    SpreadsheetApp.flush();

    return ContentService.createTextOutput(JSON.stringify({"result": "success", "id": id}))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
